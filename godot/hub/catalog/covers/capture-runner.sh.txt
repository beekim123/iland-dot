#!/usr/bin/env bash
set -euo pipefail
ROOT="$(cd "$(dirname "$0")/../.." && pwd)"
GODOT="${GODOT_BIN:-$ROOT/../tools/godot-4.7.2/Godot_v4.7.2-stable_linux.x86_64}"
OUT="$ROOT/build/catalog-capture/captures-$(date -u +%Y%m%d-%H%M%S)"
mkdir -p "$OUT"
export XDG_DATA_HOME="$OUT/isolated-user/data" XDG_CONFIG_HOME="$OUT/isolated-user/config" XDG_CACHE_HOME="$OUT/isolated-user/cache"
mkdir -p "$XDG_DATA_HOME" "$XDG_CONFIG_HOME" "$XDG_CACHE_HOME"
printf '%s\n' "$OUT" > "$ROOT/build/catalog-capture/LATEST.txt"
"$GODOT" --path "$ROOT/godot" --audio-driver Dummy --resolution 1280x800 --script "$ROOT/build/catalog-capture/capture.gd" -- --capture-output="$OUT" 2>&1 | tee "$OUT/native.log"
if grep -Eq 'SCRIPT ERROR:|Parse Error:|^ERROR:|ObjectDB.*leaked|resources still in use|CAPTURE FAIL' "$OUT/native.log"; then
 echo "Capture requires review: engine/runtime errors found"; exit 1
fi
printf 'Native capture candidates: %s\n' "$OUT"
